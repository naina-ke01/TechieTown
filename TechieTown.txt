function authMiddleware(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader)
    return res.status(401).json({ message: "Authorization header missing" });

  const token = authHeader.split(" ")[1];
  if (!token) return res.status(401).json({ message: "Token missing" });

  try {
    const decoded = jwt.verify(token, JWT_SECRET);
    req.user = decoded;
    next();
  } catch {
    return res.status(401).json({ message: "Invalid or expired token" });
  }
}

export const protect = async (req, res, next) => {
  let token;

  const authHeader = req.headers.authorization;

  if (authHeader && authHeader.startsWith("Bearer ")) {
    token = authHeader.split(" ")[1];
  }

  if (!token) {
    return res.status(401).json({ error: "Not authorized, token missing" });
  }

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    req.user = await User.findById(decoded.id).select("-password");
    if (!req.user) {
      return res.status(401).json({ error: "User not found" });
    }

    next();
  } catch (err) {
    return res.status(401).json({ error: "Not authorized, token failed" });
  }
};

    next();
  } catch (err) {
    next(err);
  }
});

router.post("/login", async (req, res) => {
  try {
    const { email, password } = req.body;

    if (!email || !password) {
      return res.status(400).json({ error: "Email and password required" });
    }

    const user = await User.findOne({ email });
    if (!user) {
      return res.status(400).json({ error: "Invalid credentials" });
    }

    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) {
      return res.status(400).json({ error: "Invalid credentials" });
    }

    const token = jwt.sign(
      { id: user._id, role: user.role },
      process.env.JWT_SECRET,
      { expiresIn: "7d" }
    );

    res.json({
      token,
      user: {
        id: user._id,
        name: user.name,
        email: user.email,
        role: user.role,
      },
    });
  } catch (err) {
    console.error("Login Error:", err);
    res.status(500).json({ error: "Login failed" });
  }
});

// Public endpoint to receive payment callbacks

  try {
    console.log("📩 M-Pesa Callback:", req.body);
    const callback = req.body?.Body?.stkCallback;

    if (!callback)
      return res.status(400).json({ error: "Invalid callback structure" });

    const {
      MerchantRequestID,
      CheckoutRequestID,
      ResultCode,
      ResultDesc,
      CallbackMetadata,
    } = callback;

    const mpesaData = {
      merchantRequestId: MerchantRequestID,
      checkoutRequestId: CheckoutRequestID,
      resultCode: ResultCode,
      resultDesc: ResultDesc,
      status: ResultCode === 0 ? "success" : "failed",
      callbackData: callback,
    };

    // If successful, extract more detailed transaction info
    if (ResultCode === 0 && CallbackMetadata) {
      const items = CallbackMetadata.Item.reduce((acc, item) => {
        acc[item.Name] = item.Value;
        return acc;
      }, {});
      mpesaData.mpesaReceiptNumber = items.MpesaReceiptNumber || "";
      mpesaData.transactionDate = items.TransactionDate || "";
      mpesaData.phone = items.PhoneNumber?.toString() || "";
      mpesaData.amount = items.Amount || 0;
    }

    // Update existing payment by CheckoutRequestID or create new one
    const payment = await Payment.findOneAndUpdate(
      { checkoutRequestId: CheckoutRequestID },
      mpesaData,
      { upsert: true, new: true }
    );

    console.log("✅ Payment updated:", payment._id);
    res.json({ success: true, message: "Callback logged", payment });
  } catch (err) {
    console.error("❌ Error logging callback:", err.message);

  const { phone, amount } = value;

  try {

    const timestamp = new Date()
      .toLocaleString("sv-SE", { timeZone: "Africa/Nairobi" })
      .replace(/\D/g, "");

    const password = Buffer.from(`${SHORTCODE}${PASSKEY}${timestamp}`).toString("base64");

    const formattedPhone = formatPhoneNumber(phone);

    const payload = {
      BusinessShortCode: SHORTCODE,
      Password: password,
      Timestamp: timestamp,
      TransactionType: "CustomerPayBillOnline",
      Amount: amount,
      PartyA: formattedPhone,
      PartyB: SHORTCODE,
      PhoneNumber: formattedPhone,
      CallBackURL: `${NGROK_URL}/api/payments/stkpush/callback`,
      AccountReference: "TechieTown",
      TransactionDesc: "Payment for TechieTown Order",
    };

    const { data } = await axios.post(
      `${baseURL}/mpesa/stkpush/v1/processrequest`,
      payload,
      { headers: { Authorization: `Bearer ${token}` } }
    );

    logger.info("✅ STK Push API Response: " + JSON.stringify(data, null, 2));

    if (data.ResponseCode === "0") {
      return res.json({
        success: true,
        message: "STK Push request accepted. Check your phone!",

    } else {
      return res.status(400).json({
        success: false,
        message: data.errorMessage || "M-Pesa rejected request",
        data,
      });
    }
  } catch (err) {
    const details = err.response?.data || err.message;
    logger.error("❌ STK Push Error: " + JSON.stringify(details, null, 2));

    res.status(500).json({
      success: false,

import jwt from "jsonwebtoken";
import User from "../models/userModel.js";

// === 🔒 Protect Middleware ===
// Checks for a valid JWT and attaches the user to req.user
export const protect = async (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({ error: "Not authorized, token missing" });
    }

    const token = authHeader.split(" ")[1];
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    // Find the user and exclude password field
    const user = await User.findById(decoded.id).select("-password");
    if (!user) {
      return res.status(401).json({ error: "User not found" });
    }

    req.user = user;
    next();
  } catch (err) {
    console.error("❌ Auth Error:", err.message);
    res.status(401).json({ error: "Not authorized, invalid or expired token" });
  }
};

// === 🛡️ Admin Only Middleware ===
// Ensures the logged-in user has admin privileges
export const adminOnly = (req, res, next) => {
  if (!req.user) {
    return res.status(401).json({ error: "Not authorized" });
  }

  if (req.user.role !== "admin") {
    return res.status(403).json({ error: "Admin access required" });
  }

  next();
};

/**
 * @desc  Initiate STK Push Payment
 * @route POST /api/mpesa/stkpush
 */
export const stkPush = async (req, res) => {
  try {
    const { phone, amount } = req.body;

    if (!phone || !amount) {
      return res.status(400).json({ error: "Phone and amount are required" });
    }

    const token = await getAccessToken(CONSUMER_KEY, CONSUMER_SECRET, baseURL);

    const timestamp = new Date()
      .toLocaleString("sv-SE", { timeZone: "Africa/Nairobi" })
      .replace(/\D/g, "");

    const password = Buffer.from(`${SHORTCODE}${PASSKEY}${timestamp}`).toString("base64");

    const payload = {

      Password: password,
      Timestamp: timestamp,
      TransactionType: "CustomerPayBillOnline",
      Amount: amount,
      PartyA: formatPhoneNumber(phone),
      PartyB: SHORTCODE,
      PhoneNumber: formatPhoneNumber(phone),
      CallBackURL: callbackURL,
      AccountReference: "TechieTown",
      TransactionDesc: "Payment for TechieTown Order",
    };

    const { data } = await axios.post(
      `${baseURL}/mpesa/stkpush/v1/processrequest`,
      payload,
      { headers: { Authorization: `Bearer ${token}` } }
    );

    // Store initial request in DB
    await Payment.create({
      phone: formatPhoneNumber(phone),
      amount,
      CheckoutRequestID: data.CheckoutRequestID,
      MerchantRequestID: data.MerchantRequestID,
      status: "Pending",
      rawResponse: data,
    });

    logger.info(`✅ STK Push initiated for ${phone}: ${JSON.stringify(data)}`);
    res.json({ success: true, message: "STK Push initiated", data });

import axios from "axios";

/**
 * ✅ Formats a phone number into the required Safaricom format: 2547XXXXXXXX
 * Accepts numbers like:
 *  - 07XXXXXXXX
 *  - 7XXXXXXXX
 *  - 2547XXXXXXXX
 *  - +2547XXXXXXXX
 */
export function formatPhoneNumber(phone) {
  if (!phone) throw new Error("Phone number is required");

  let formatted = phone.toString().trim().replace(/\s+/g, "");

  // Remove '+' if present
  if (formatted.startsWith("+")) {
    formatted = formatted.slice(1);
  }

  // If number starts with 07..., replace with 2547...
  if (formatted.startsWith("07")) {
    formatted = "254" + formatted.substring(1);
  }

  // If number starts with just 7..., prepend 254
  if (formatted.startsWith("7")) {
    formatted = "254" + formatted;
  }

  // Ensure 254 prefix
  if (!formatted.startsWith("254")) {
    throw new Error(`Invalid phone format: ${phone}`);
  }

  return formatted;
}

/**
 * ✅ Fetches M-Pesa access token
 */
export async function getAccessToken(CONSUMER_KEY, CONSUMER_SECRET, baseURL) {
  try {
    const auth = Buffer.from(`${CONSUMER_KEY}:${CONSUMER_SECRET}`).toString("base64");

    const { data } = await axios.get(
      `${baseURL}/oauth/v1/generate?grant_type=client_credentials`,
      {
        headers: { Authorization: `Basic ${auth}` },
      }
    );

    if (!data.access_token) throw new Error("No access token received from M-Pesa");

    return data.access_token;
  } catch (err) {
    console.error("❌ Access Token Error:", err.response?.data || err.message);
    throw new Error("Failed to get M-Pesa access token");
  }
}

document.addEventListener("DOMContentLoaded", async () => {
  const loginForm = document.getElementById("loginForm");
  const loginBtn = document.getElementById("loginBtn");
  const message = document.getElementById("message");

  if (!loginForm || !loginBtn) return;

  // === STEP 1: Fetch backend base URL dynamically ===
  try {
    // If frontend served via backend (localhost/ngrok)
    const res = await fetch("/api/config");
    const data = await res.json();

    // Fallbacks for local or file://
    API_BASE_URL =
      data.baseUrl ||
      window.location.origin ||
      "http://localhost:3000";

    console.log("✅ Backend Base URL:", API_BASE_URL);
  } catch (err) {
    // Fallback for file:// or offline mode
    console.warn("⚠️ Failed to fetch backend config, using localhost fallback:", err);
    API_BASE_URL = "http://localhost:3000";
  }

  // === Helper: Decode JWT ===
  function parseJwt(token) {
    try {
      const base64Url = token.split(".")[1];
      const base64 = base64Url.replace(/-/g, "+").replace(/_/g, "/");
      const jsonPayload = decodeURIComponent(
        atob(base64)
          .split("")
          .map((c) => "%" + ("00" + c.charCodeAt(0).toString(16)).slice(-2))
          .join("")
      );
      return JSON.parse(jsonPayload);
    } catch {
      return null;
    }
  }

  // === STEP 2: If already logged in, redirect to dashboard ===
  const existingToken = localStorage.getItem("adminToken");
  const existingUser = JSON.parse(localStorage.getItem("adminUser") || "{}");

  if (existingToken && existingUser.role === "admin") {
    console.log("🔁 Already logged in. Redirecting to dashboard...");
    window.location.href = "/admin/a.dashboard.html";
    return;
  }

  // === STEP 3: Handle Login ===
  loginForm.addEventListener("submit", async (e) => {
    e.preventDefault();

    const email = document.getElementById("email").value.trim();
    const password = document.getElementById("password").value.trim();

    message.textContent = "";
    message.className = "";

    if (!email || !password) {
      message.textContent = "Please enter both email and password.";
      message.className = "error";
      return;
    }

    if (!navigator.onLine) {
      message.textContent = "You are offline. Check your internet connection.";
      message.className = "error";
      return;
    }

    loginBtn.disabled = true;
    loginBtn.textContent = "Logging in...";

    try {
      const res = await fetch(`${API_BASE_URL}/api/users/login`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, password }),
      });

      let data;
      try {
        data = await res.json();
      } catch {
        throw new Error("Invalid response format (not JSON)");
      }

      if (!res.ok) {
        message.textContent =
          data.error || data.message || "Invalid login credentials.";
        message.className = "error";
        return;
      }

      const { token, user } = data;
      const decoded = parseJwt(token);

      if (!decoded || decoded.role !== "admin" || !user || user.role !== "admin") {
        message.textContent = "Access denied: Admins only.";
        message.className = "error";
        return;
      }

      // ✅ Store token & user info
      localStorage.setItem("adminToken", token);
      localStorage.setItem("adminUser", JSON.stringify(user));
      console.log("💾 Token stored:", token);
      console.log("💾 User stored:", user);

      message.className = "success";

      message.textContent = "Server error. Please try again later.";
      message.className = "error";
    } finally {
      loginBtn.disabled = false;
      loginBtn.textContent = "Login";
    }
  });

import fs from "fs/promises";
import path from "path";

export async function appendToJSONFile(filePath, data) {
  try {
    const dir = path.dirname(filePath);
    await fs.mkdir(dir, { recursive: true });

    let existing = [];
    try {
      const fileContent = await fs.readFile(filePath, "utf8");
      existing = JSON.parse(fileContent || "[]");
    } catch (err) {
      if (err.code !== "ENOENT") throw err;
    }

    existing.push(data);
    await fs.writeFile(filePath, JSON.stringify(existing, null, 2), "utf8");
  } catch (err) {
    throw new Error("File write error: " + err.message);
  }
}

app.post("/api/payments/stkpush", async (req, res) => {
  const { error, value } = stkPushSchema.validate(req.body);
  if (error) {
    return res.status(400).json({
      success: false,
      message: error.details[0].message,
    });
  }

  const { phone, amount } = value;

  try {
    const token = await getAccessToken(CONSUMER_KEY, CONSUMER_SECRET, baseURL);

    const timestamp = new Date()
      .toLocaleString("sv-SE", { timeZone: "Africa/Nairobi" })
      .replace(/\D/g, "");

    const password = Buffer.from(`${SHORTCODE}${PASSKEY}${timestamp}`).toString("base64");

    const formattedPhone = formatPhoneNumber(phone);

    const payload = {
      BusinessShortCode: SHORTCODE,
      Password: password,
      Timestamp: timestamp,
      TransactionType: "CustomerPayBillOnline",
      Amount: amount,
      PartyA: formattedPhone,
      PartyB: SHORTCODE,
      PhoneNumber: formattedPhone,
      CallBackURL: `${NGROK_URL}/api/payments/stkpush/callback`,
      AccountReference: "TechieTown",
      TransactionDesc: "Payment for TechieTown Order",
    };

    const { data } = await axios.post(
      `${baseURL}/mpesa/stkpush/v1/processrequest`,
      payload,
      { headers: { Authorization: `Bearer ${token}` } }
    );

    logger.info("✅ STK Push API Response: " + JSON.stringify(data, null, 2));

    if (data.ResponseCode === "0") {
      return res.json({
        success: true,
        message: "STK Push request accepted. Check your phone!",
        data,
      });
    } else {
      return res.status(400).json({
        success: false,
        message: data.errorMessage || "M-Pesa rejected request",
        data,
      });
    }
  } catch (err) {
    const details = err.response?.data || err.message;
    logger.error("❌ STK Push Error: " + JSON.stringify(details, null, 2));

    res.status(500).json({
      success: false,
      message: "Failed to initiate payment",
      details,
    });
  }
});

} = process.env;

const baseURL =

/**
 * @desc  Initiate STK Push Payment
 * @route POST /api/mpesa/stkpush
 */
export const stkPush = async (req, res) => {
  try {
    const { phone, amount } = req.body;

    if (!phone || !amount) {
      return res.status(400).json({ error: "Phone and amount are required" });
    }

    const token = await getAccessToken(CONSUMER_KEY, CONSUMER_SECRET, baseURL);

    const timestamp = new Date()
      .toLocaleString("sv-SE", { timeZone: "Africa/Nairobi" })
      .replace(/\D/g, "");

    const password = Buffer.from(`${SHORTCODE}${PASSKEY}${timestamp}`).toString("base64");

    const payload = {
      BusinessShortCode: SHORTCODE,
      Password: password,
      Timestamp: timestamp,
      TransactionType: "CustomerPayBillOnline",
      Amount: amount,
      PartyA: formatPhoneNumber(phone),
      PartyB: SHORTCODE,
      PhoneNumber: formatPhoneNumber(phone),
      CallBackURL: callbackURL,
      AccountReference: "TechieTown",
      TransactionDesc: "Payment for TechieTown Order",
    };

    const { data } = await axios.post(
      `${baseURL}/mpesa/stkpush/v1/processrequest`,
      payload,
      { headers: { Authorization: `Bearer ${token}` } }
    );

    // Store initial request in DB
    await Payment.create({
      phone: formatPhoneNumber(phone),
      amount,
      CheckoutRequestID: data.CheckoutRequestID,
      MerchantRequestID: data.MerchantRequestID,
      status: "Pending",
      rawResponse: data,
    });

    logger.info(`✅ STK Push initiated for ${phone}: ${JSON.stringify(data)}`);
    res.json({ success: true, message: "STK Push initiated", data });
  } catch (err) {
    const details = err.response?.data || err.message;
    logger.error(`❌ STK Push Error: ${JSON.stringify(details)}`);
    res.status(500).json({ error: "Failed to initiate payment", details });
  }
};

/**
 * @desc  Handle M-PESA Callback
 * @route POST /api/mpesa/callback
 */
export const mpesaCallback = async (req, res) => {
  try {
    const callback = req.body.Body?.stkCallback;
    if (!callback) {
      logger.warn("⚠️ Empty M-PESA callback received");
      return res.status(400).json({ message: "Invalid callback format" });
    }

    const { ResultCode, ResultDesc, CheckoutRequestID, CallbackMetadata } = callback;

    const payment = await Payment.findOneAndUpdate(
      { CheckoutRequestID },
      {
        status: ResultCode === 0 ? "Success" : "Failed",
        ResultCode,
        ResultDesc,
        metadata: CallbackMetadata?.Item || [],
      },
      { new: true }
    );

    if (!payment) {
      logger.warn(`⚠️ No payment found for CheckoutRequestID: ${CheckoutRequestID}`);
    } else {
      logger.info(`📩 Callback processed: ${ResultDesc}`);
    }

    res.json({ message: "Callback received successfully" });
  } catch (err) {
    logger.error(`❌ Callback processing error: ${err.message}`);
    res.status(500).json({ error: "Failed to process callback" });
  }
};

import User from "../models/userModel.js";
import asyncHandler from "express-async-handler";
import jwt from "jsonwebtoken";

// Generate JWT with id and role
const generateToken = (user) => {
  return jwt.sign(
    { id: user._id, role: user.role },
    process.env.JWT_SECRET,
    { expiresIn: "7d" }
  );
};

// @desc Auth user & get token
// @route POST /api/users/login
// @access Public
export const authUser = asyncHandler(async (req, res) => {
  const { email, password } = req.body;

  const user = await User.findOne({ email });
  if (!user) {
    res.status(401);
    throw new Error("Invalid email or password");
  }

  const isMatch = await user.matchPassword(password);
  if (!isMatch) {
    res.status(401);
    throw new Error("Invalid email or password");
  }

  // Return token + user info including role
  res.json({
    user: {
      id: user._id,
      name: user.name,
      email: user.email,
      role: user.role,
    },
    token: generateToken(user),
  });
});



public/js/adminlogin.js
let API_BASE_URL = null;

document.addEventListener("DOMContentLoaded", async () => {
  const loginForm = document.getElementById("loginForm");
  const loginBtn = document.getElementById("loginBtn");
  const message = document.getElementById("message");

  if (!loginForm || !loginBtn) return;

  // === STEP 1: Fetch backend base URL dynamically ===
  try {
    // If frontend served via backend (localhost/ngrok)
    const res = await fetch("/api/config");
    const data = await res.json();

    // Fallbacks for local or file://
    API_BASE_URL =
      data.baseUrl ||
      window.location.origin ||
      "http://localhost:3000";

    console.log("✅ Backend Base URL:", API_BASE_URL);
  } catch (err) {
    // Fallback for file:// or offline mode
    console.warn("⚠️ Failed to fetch backend config, using localhost fallback:", err);
    API_BASE_URL = "http://localhost:3000";
  }

  // === Helper: Decode JWT ===
  function parseJwt(token) {
    try {
      const base64Url = token.split(".")[1];
      const base64 = base64Url.replace(/-/g, "+").replace(/_/g, "/");
      const jsonPayload = decodeURIComponent(
        atob(base64)
          .split("")
          .map((c) => "%" + ("00" + c.charCodeAt(0).toString(16)).slice(-2))
          .join("")
      );
      return JSON.parse(jsonPayload);
    } catch {
      return null;
    }
  }

  // === STEP 2: If already logged in, redirect to dashboard ===
  const existingToken = localStorage.getItem("adminToken");
  const existingUser = JSON.parse(localStorage.getItem("adminUser") || "{}");

  if (existingToken && existingUser.role === "admin") {
    console.log("🔁 Already logged in. Redirecting to dashboard...");
    window.location.href = "/admin/a.dashboard.html";
    return;
  }

  // === STEP 3: Handle Login ===
  loginForm.addEventListener("submit", async (e) => {
    e.preventDefault();

    const email = document.getElementById("email").value.trim();
    const password = document.getElementById("password").value.trim();

    message.textContent = "";
    message.className = "";

    if (!email || !password) {
      message.textContent = "Please enter both email and password.";
      message.className = "error";
      return;
    }

    if (!navigator.onLine) {
      message.textContent = "You are offline. Check your internet connection.";
      message.className = "error";
      return;
    }

    loginBtn.disabled = true;
    loginBtn.textContent = "Logging in...";

    try {
      const res = await fetch(`${API_BASE_URL}/api/users/login`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, password }),
      });

      let data;
      try {
        data = await res.json();
      } catch {
        throw new Error("Invalid response format (not JSON)");
      }

      if (!res.ok) {
        message.textContent =
          data.error || data.message || "Invalid login credentials.";
        message.className = "error";
        return;
      }

      const { token, user } = data;
      const decoded = parseJwt(token);

      if (!decoded || decoded.role !== "admin" || !user || user.role !== "admin") {
        message.textContent = "Access denied: Admins only.";
        message.className = "error";
        return;
      }

      // ✅ Store token & user info
      localStorage.setItem("adminToken", token);
      localStorage.setItem("adminUser", JSON.stringify(user));
      console.log("💾 Token stored:", token);
      console.log("💾 User stored:", user);

      message.textContent = "Login successful! Redirecting...";
      message.className = "success";

      setTimeout(() => {
        window.location.href = "/admin/a.dashboard.html";
      }, 1000);
    } catch (err) {
      console.error("❌ Login Error:", err);
      message.textContent = "Server error. Please try again later.";
      message.className = "error";
    } finally {
      loginBtn.disabled = false;
      loginBtn.textContent = "Login";
    }
  });
});

Perfect — thanks for sharing the full TechieTown structure. That actually makes placing the /api/payments/stkpush endpoint very clear. Here’s exactly where everything goes and how it should be organized:

1️⃣ Place the STK push route

You already have a routes folder:

backend/routes/
├─ paymentRoute.js      (existing)
├─ mpesaRoute.js        (existing)


Option A (cleanest):

Put the STK push endpoint in mpesaRoute.js because it’s specifically M-Pesa logic.

Or, if you want to keep M-Pesa routes separate from other payments, create stkpushRoute.js, but it’s fine inside mpesaRoute.js.

2️⃣ Place the logic for STK push

You have a utils folder:

backend/utils/
├─ mpesa.js


✅ Ideal: Put the functions for:

getAccessToken()

sendSTKPush()

getTimestamp()

generatePassword()

inside mpesa.js as reusable functions. Then import them in your route:

import express from 'express';
import { getAccessToken, sendSTKPush } from '../utils/mpesa.js';

const router = express.Router();

router.post('/stkpush', async (req, res) => {
    try {
        const { phone, amount } = req.body;
        if (!phone || !amount) return res.status(400).json({ error: 'Phone and amount required' });

        const stkRes = await sendSTKPush(phone, amount); // call utility function
        res.status(200).json(stkRes);
    } catch (err) {
        console.error(err);
        res.status(500).json({ error: 'STK Push failed', details: err.message });
    }
});

export default router;
